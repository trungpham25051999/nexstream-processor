cmake_minimum_required(VERSION 3.20)
project(nexstream_processor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NEXSTREAM_ENABLE_SANITIZERS "ASan/UBSan" OFF)
include(CTest)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(nexstream_processor_core src/domain/evaluator.cpp src/infrastructure/json_codec.cpp src/application/partition_state.cpp)
target_include_directories(nexstream_processor_core PUBLIC include PRIVATE ${JSON_INCLUDE_DIR})
target_compile_options(nexstream_processor_core PRIVATE -Wall -Wextra -Wpedantic)
if(NEXSTREAM_ENABLE_SANITIZERS)
target_compile_options(nexstream_processor_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(nexstream_processor_core PUBLIC -fsanitize=address,undefined)
endif()
